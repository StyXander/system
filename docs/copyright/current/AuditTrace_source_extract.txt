@app.middleware('http')
async def connectivity_timing(request, call_next):
    started = time.perf_counter()
    response = await call_next(request)
    if request.url.path.startswith('/api/connectivity/runs'):
        duration = (time.perf_counter() - started) * 1000
        response.headers['Server-Timing'] = f'app;dur={duration:.2f}'
        logging.getLogger('uvicorn.error').info('connectivity_http method=%s path=%s status=%s duration_ms=%.2f 
bytes=%s',
            request.method, request.url.path, response.status_code, duration, response.headers.get('content-leng
th', 'stream'))
    return response

def build_router(root: Path, reserve_model, quota_ledger) -> APIRouter:
    """延迟绑定现有应用配额，避免循环导入和第二套模型预算。"""
    router = APIRouter(prefix="/api/connectivity", tags=["connectivity"])
    lock = threading.RLock()
    stores = {}
    directory = root / "backend/runtime/connectivity"
    directory.mkdir(parents=True, exist_ok=True)
    result_cache = OrderedDict()
    log = logging.getLogger(__name__)

    def normalized_result(item, *, layer="full"):
        body = item.get("result")
        if not body or body.get("projection_supported") is False:
            return body
        from .connectivity_extraction import source_path
        from .connectivity_source_cache import source_signature
        registry = {d.document_id: d for d in load_registry(root)}
        signatures = []
        for original in body.get("documents", []):
            doc = registry.get(original["document_id"])
            try:
                signatures.append((original["document_id"], doc.file_sha256, source_signature(source_path(root, 
doc))))
            except (OSError, AttributeError, ValueError):
                signatures.append((original["document_id"], "missing"))
        rules = root / "backend/app/connectivity_matter_rules.json"
        try:
            rule_signature = source_signature(rules)
        except OSError:
            rule_signature = None
        version = item.get("version", item.get("updated_at"))
        key = (item["task_id"], version, layer, tuple(signatures), rule_signature)
        with lock:
            if key in result_cache:
                result_cache.move_to_end(key)
                log.info("connectivity_result cache=hit task=%s layer=%s version=%s", item["task_id"], layer, ve
rsion)
                return deepcopy(result_cache[key])
        normalized = ConnectivityRun.model_validate({k: v for k, v in body.items() if k in ConnectivityRun.model
_fields})
        normalized = refresh_run_sources(root, normalized)
        if layer == "full":
            if normalized.matter_schema_version:
                from .connectivity_assurance_planning import build_assurance_planning
                normalized.assurance_planning = build_assurance_planning(normalized)
            value = normalized.model_dump(mode="json")
            value["overview"] = build_overview(normalized)
        else:
            all_fields = normalized.model_dump(mode="json")
            value = {k: all_fields.get(k, v) for k, v in body.items()}
            value["versions"] = all_fields["versions"]
            if layer in {"summary", "matters"} and normalized.versions.get("source_validity", {}).get("status") 
== "invalid":
                # 摘要没有证据全文，不能继续展示来源有效的旧判断。
                value["assurance_planning"] = {}
                for assessment in value.get("matter_assessments", []):
                    assessment["relation_support"] = "invalid"
                    assessment["unresolved"] = [*assessment.get("unresolved", []), "依赖来源变化，打开事项查看失
效证据"]
        value["ai_generated_content_notice"] = NOTICE
        with lock:
            # 版本变化即淘汰旧版本；来源变化也不积累大结果。
            for old in list(result_cache):
                if old[0] == item["task_id"] and (old[1] != version or old[2] == layer):
                    del result_cache[old]
            result_cache[key] = deepcopy(value)
            while len(result_cache) > 16:
                result_cache.popitem(last=False)
        log.info("connectivity_result cache=miss task=%s layer=%s version=%s", item["task_id"], layer, version)
        return value

    def store(task_id=None):
        mode = "local" if task_id and task_id.startswith("CONNJOB-") else ("supabase" if task_id or demo_task_su
pabase_enabled() else "local")
        with lock:
            if mode not in stores:
                stores[mode] = SupabaseDemoRunTaskStore(get_demo_task_client()) if mode == "supabase" else DemoR
unTaskStore(directory / "tasks", task_id_prefix="CONNJOB")
        return stores[mode]

    def owner(request: Request) -> str:
        return hashlib.sha256(request.cookies.get(COOKIE, "").encode()).hexdigest()

    def task(task_id: str):
        result = store(task_id).get(task_id)
        if not result or result.get("run_body", {}).get("kind") != "connectivity":
            raise HTTPException(404, "关联核对任务不存在")
        return result

    def authorize(request: Request, item: dict):
        if not single_user_enabled() and (not request.cookies.get(COOKIE) or not hmac.compare_digest(owner(reque
st), item["run_body"].get("owner", ""))):
            raise HTTPException(403, "只能更正或取消当前浏览器创建的任务")
        origin = request.headers.get("origin")
        if origin and origin.rstrip("/") != str(request.base_url).rstrip("/"):
            raise HTTPException(403, "更正请求必须来自当前网站")

    def public(item: dict, request: Request | None = None) -> dict:
        view = store(item["task_id"]).snapshot(item["task_id"], item=item)
        if not view:
            raise HTTPException(404, "任务不存在")
        result = {k: v for k, v in view.items() if k not in {"run_body", "request_payload", "request_sha256", "i
dempotency_key_sha256"}}
        result["parent_task_id"] = item.get("run_body", {}).get("parent_task_id")
        is_owner = single_user_enabled() or bool(request and request.cookies.get(COOKIE)
                        and hmac.compare_digest(owner(request), item.get("run_body", {}).get("owner", "")))
        if item.get("run_body", {}).get("analysis_scope") == "matter_connectivity" and not is_owner:
            raise HTTPException(403, "事项调查任务仅当前创建会话可读取")
        result["can_modify"] = is_owner
        # 公共视图默认不含个人观察正文，只列条数；本人视图才展开。
        body = result.get("result")
        if isinstance(body, dict):
            body = normalized_result({**item, "result": body})
            result['result'] = body
        if isinstance(body, dict) and not is_owner:
            body = dict(body)
            notes = body.get("professional_notes") or []
            body["personal_observation_count"] = len(notes)
            body["professional_notes"] = []
            result["result"] = body
        elif isinstance(body, dict):
            body = dict(body)
            body["personal_observation_count"] = len(body.get("professional_notes") or [])
            result["result"] = body
        return result

    @router.post("/runs/{task_id}/reviews")
    def review(task_id: str, body: UserReviewRequest, request: Request):
        item, run = readable(task_id)
        authorize(request, item)
        if body.expected_revision != run.result_revision:
            raise HTTPException(409, "结果版本已变化，请重新读取")
        ids = set(body.check_ids)
        if len(ids) != len(body.check_ids) or not ids.issubset({c.check_id for c in run.checks}):
            raise HTTPException(422, "所选关系必须属于本次结果且不可重复")
        if body.action == "personal_observation" and (len(ids) != 1 or len(body.observation.strip()) < 3):
            raise HTTPException(422, "个人观察须对应一项关系并填写具体内容")
        if body.action in {"ai_accept", "ai_reject"}:
            if len(ids) != 1 or not any(a.check_id in ids and a.status == "current" for a in run.ai_analyses):
                raise HTTPException(422, "请选一项与当前结果绑定的AI意见；旧意见需先重新分析")
        child = run.model_copy(deep=True)
        child.parent_run_id = run.run_id
        child.run_id = f"CONN-{uuid.uuid4().hex[:16].upper()}"
        child.result_revision += 1
        for check in child.checks:
            if check.check_id not in ids:
                continue
            before = check.model_dump()
            if body.action in {"ai_accept", "ai_reject"}:
                opinion = next(a for a in child.ai_analyses if a.check_id == check.check_id)
                before = opinion.model_dump()
                opinion.review_status = "accepted" if body.action == "ai_accept" else "rejected"
                child.reviews.append(ReviewEvent(event_id=body.action_id, actor=body.actor, action=body.action,
                    reason=body.reason, check_id=check.check_id, before=before, after=opinion.model_dump()))
                continue
            elif body.action == "reading_confirmation":
                check.review_status = "read_confirmed"
            else:
                child.professional_notes.append(ProfessionalRiskNote(check_id=check.check_id,
                    observation=body.observation.strip(), reason=body.reason, actor=body.actor))
            child.reviews.append(ReviewEvent(event_id=body.action_id, actor=body.actor, action=body.action,
                reason=body.reason, check_id=check.check_id, before=before, after=check.model_dump()))
        return save_revision(item, run, body, request, child)

    def save_revision(item, run, body, request, child):
        if os.getenv("AUDITTRACE_DEMO_EXECUTOR_MODE","web").strip().lower() != "web":
            raise HTTPException(503,"关联核对修订执行器未就绪：当前版本须配置Web执行模式")
        runtime = store(item["task_id"])
        mutation_digest = hashlib.sha256(body.model_dump_json().encode()).hexdigest()
        with lock:
            head_file = directory / "revisions" / f"{run.run_id}.json"
            head_file.parent.mkdir(parents=True, exist_ok=True)
            remote = not item["task_id"].startswith("CONNJOB-")
            if remote:
                # 子修订本身也在共享任务表内，刷新或实例更换不依赖本机侧文件。

                    params={"request_payload->>parent_run_id": f"eq.{run.run_id}", "task_kind": "eq.connectivity
", "select": "task_id,request_payload", "limit": "1"})
                if rows:
                    previous = rows[0]
                    if previous["request_payload"].get("action_id") == body.action_id:
                        if previous["request_payload"].get("mutation_digest") != mutation_digest:
                            raise HTTPException(409, "同一操作编号不能提交不同内容")
                        return public(task(previous["task_id"]), request)
                    raise HTTPException(409, "该结果已有新修订，请读取子运行后再更正")
            elif head_file.exists():
                head = json.loads(head_file.read_text(encoding="utf-8"))
                if head["action_id"] == body.action_id:
                    if head.get("mutation_digest") != mutation_digest:
                        raise HTTPException(409, "同一操作编号不能提交不同内容")
                    return public(task(head["task_id"]), request)
                raise HTTPException(409, "该结果已有新修订，请读取子运行后再更正")
            def save_child(row):
                if child.matter_schema_version:
                    from .connectivity_matters import finish_projection
                    finish_projection(child)
                refresh_analysis_status(child)
                child.versions['result_origin'] = 'user_revision'
                child.program_hints = build_program_hints(child)
                child.overview = build_overview(child)
                runtime.update_task(row["task_id"], status="completed", run_id=child.run_id, result=child.model_
dump(mode="json"), result_expires_at=result_expiry_iso(minimum_seconds=CONNECTIVITY_RETENTION_SECONDS))
            correction_body = {"kind": "connectivity", "owner": owner(request), "analysis_scope": run.versions.g
et("analysis_scope", "legacy_checks"), "parent_run_id": run.run_id, "parent_task_id": item["task_id"], "action_i
d": body.action_id, "mutation_digest": mutation_digest}
            try:
                # 每个父版本使用一个稳定修订键，复用共享任务表既有唯一约束。
                result = runtime.create(item["case_id"], correction_body, save_child, idempotency_key=f"connecti
vity-revision-{run.run_id}")
            except (SupabaseConflict, IdempotencyConflict) as error:
                raise HTTPException(409, "修订发生并发冲突，请重新读取当前结果") from error
            if not remote:
                temporary = head_file.with_suffix(".tmp")

    if isinstance(value, AgentMatterResult):
        # 供应商适配器已经转成统一对象；不能将别的角色的非空内容带入。
        value = {k: v for k, v in value.model_dump().items() if k in schema.model_fields or v not in (None, [], 
"")}
    result = schema.model_validate(value)
    if not combined and not result.tool_requests:
        if role == "A1" and not result.matters:
            raise ValueError("matters: 最终建档必须明确所选事项身份，无法确认应保留候选并写依据")

            raise ValueError("assessment: 综合判断缺失；证据不足也必须保留缺口判断")
    return AgentMatterResult.model_validate(result.model_dump())

"""原件哈希缓存。每次检查文件身份，最终导出可强制重读字节。"""
from collections import OrderedDict
import hashlib
import logging
from pathlib import Path
import threading
import time
from weakref import WeakValueDictionary

_lock = threading.RLock()
_cache = OrderedDict()
_file_locks = WeakValueDictionary()
_MAX_ENTRIES = 512
log = logging.getLogger(__name__)


def source_signature(path):
    path = Path(path).resolve()
    stat = path.stat()
    return (str(path), stat.st_size, stat.st_mtime_ns)


def source_sha256(path, *, force=False):
    # 同一文件并发读取合并；读取期间文件变化时拒绝缓存不稳定内容。
    with _lock:
        name = str(Path(path).resolve())
        file_lock = _file_locks.get(name)
        if file_lock is None:
            file_lock = threading.RLock()
            _file_locks[name] = file_lock
    with file_lock:
        key = source_signature(path)
        with _lock:
            if not force and key in _cache:
                _cache.move_to_end(key)
                log.info("source_hash cache=hit bytes=%s", key[1])
                return _cache[key]
        started = time.perf_counter()
        digest = hashlib.sha256()
        with open(path, "rb") as handle:
            for chunk in iter(lambda: handle.read(1024 * 1024), b""):
                digest.update(chunk)
        if source_signature(path) != key:
            raise OSError("来源在校验期间发生变化")
        # 同路径只保留最新指纹，避免文件换版累积缓存。
        result = digest.hexdigest()
        with _lock:
            for old in list(_cache):
                if old[0] == key[0]:
                    del _cache[old]
            _cache[key] = result
            while len(_cache) > _MAX_ENTRIES:
                _cache.popitem(last=False)
        log.info("source_hash cache=miss forced=%s bytes=%s duration_ms=%.2f", force, key[1], (time.perf_counter
()-started)*1000)
        return result
"""表内结构自证：合计行/列、分项行/列与必要的跨表桥接候选。

只从已定位的同一表格结构生成候选；结构证据支持的是 L1 有限可比，
不证明指标定义、统计范围、期间或业务互不重叠。是否正式闭合仍由
`upgrade_conditions` 与 `calculate` 的原文条件门禁决定。
"""
from __future__ import annotations

from collections import defaultdict
from decimal import Decimal
import hashlib

from .connectivity_engine import number, numeric_unit, text_number
from .connectivity_rules import RULES, required_conditions

    RelationshipMember)

TOTAL_LABEL = re.compile(r"(?:合计|总计|总额|小计|总投入|投入合计)$")
COMPONENT_HINT = re.compile(r"^(?:其中|资本化|费用化|专项|其他|其它|钢铁|分项|明细)")
COMPONENT_IN_LABEL = re.compile(r"其中|资本化|费用化|钢铁|其他|其它|分项|明细")
SUM_TOPICS = {"environment": "ENV-001", "research": "RD-001"}


def _position(occurrence: DisclosureOccurrence) -> dict | None:
    """表内自证只接受新提取器写入的显式表格身份；旧运行须重新提取。"""
    if not (occurrence.table_id and occurrence.row_index is not None and occurrence.column_index is not None):
        return None
    return {"table_id": occurrence.table_id, "row": occurrence.row_index, "column": occurrence.column_index,
            "page": occurrence.pdf_page, "document": occurrence.document_id}


def _usable(row: DisclosureOccurrence, period: str) -> bool:
    """组成候选只读已定位、公司自身、有明确数值和量纲的单元格。"""
    if row.value is None or row.quality != "located":
        return False
    if not row.year or row.year_status != "attributed":
        return False
    if row.content_class in {"macro", "industry", "peer", "background"}:
        return False
    if period and row.year and row.year != period:
        return False
    try:
        number(row.value)
    except ValueError:
        return False
    return numeric_unit(row.unit)[0] != "unknown"


def _normalized(row: DisclosureOccurrence) -> tuple[Decimal, str]:
    value = number(row.value)
    dimension, factor = numeric_unit(row.unit)
    return value * factor, dimension


def _reference(row: DisclosureOccurrence) -> EvidenceRef:
    quote = (row.row_header or row.context or "").strip()[:800]
    return EvidenceRef(occurrence_id=row.occurrence_id, quote=quote)


def _rows_by_table(occurrences: list[DisclosureOccurrence], period: str) -> dict[str, dict]:
    """按真实 table_id 聚合行/列位置；没有结构身份的坐标行不参与表内自证。"""
    tables: dict[str, dict] = {}
    for row in occurrences:
        if not _usable(row, period):
            continue
        position = _position(row)
        if position is None:
            continue
        table = tables.setdefault(position["table_id"], {
            "table_id": position["table_id"], "document_id": row.document_id,
            "pdf_page": position["page"], "rows": defaultdict(dict), "row_labels": {},
            "column_labels": {}, "header_paths": {},
        })
        table["rows"][position["row"]][position["column"]] = row
        table["row_labels"].setdefault(position["row"], row.label)
        table["column_labels"].setdefault(position["column"], row.column_header or " / ".join(row.header_path))
        table["header_paths"].setdefault(position["column"], list(row.header_path))
    return tables


def _role_from_label(label: str) -> str:
    compact = re.sub(r"[\s：:（）()]", "", label or "")
    if TOTAL_LABEL.search(compact):
        return "total"
    if COMPONENT_HINT.search(compact) or COMPONENT_IN_LABEL.search(compact):
        return "component"
    return "detail"


def _role_from_header(header: str) -> str:
    compact = re.sub(r"\s+", "", header or "")
    if re.search(r"合计|总计|总额|小计", compact):
        return "total"
    if re.search(r"钢铁|其他|其它|范围一|范围二|费用化|资本化|分项|明细", compact):
        return "component"
    if re.search(r"20\d{2}|本期|本年", compact):
        return "period"
    return "unknown"


def _member(row: DisclosureOccurrence, role: str) -> RelationshipMember:
    return RelationshipMember(occurrence_id=row.occurrence_id, role=role)


def _condition(name: str, reason: str, evidence: list[EvidenceRef]) -> ComparisonCondition:
    return ComparisonCondition(name=name, reason=reason, evidence=evidence,
                               evidence_kind="table_structure" if name in {"组成完整", "互不重叠"} else "unspeci
fied")


def _structure_payload(table_id: str, layout: str, total: DisclosureOccurrence,
                       components: list[DisclosureOccurrence], *, coverage: dict,
                       non_overlap: dict, limitations: list[str]) -> dict:
    """结构证据保存可复核的坐标、角色、覆盖与限制，不用“加平”代替完整性。"""
    return {
        "table_id": table_id,
        "layout": layout,
        "pdf_page": total.pdf_page,
        "header_path": list(total.header_path),
        "total": {"occurrence_id": total.occurrence_id, "label": total.label, "row_index": total.row_index,
                  "column_index": total.column_index, "row_role": "total",
                  "column_role": total.column_role if total.column_role != "unknown" else "period",
                  "column_header": total.column_header, "header_path": list(total.header_path),
                  "raw": total.value, "unit": total.unit},
        "components": [{"occurrence_id": row.occurrence_id, "label": row.label, "row_index": row.row_index,
                        "column_index": row.column_index, "row_role": row.row_role,
                        "column_role": row.column_role, "column_header": row.column_header,
                        "header_path": list(row.header_path), "raw": row.value, "unit": row.unit} for row in com
ponents],
        "coverage": coverage,
        "non_overlap": non_overlap,
        "limitations": limitations,
    }


def _check(rule_id: str, title: str, topic: str, total: DisclosureOccurrence,
           components: list[DisclosureOccurrence], structure: dict) -> RelationshipCheck:
    required = required_conditions(RULES[rule_id]["operator"])
    reasons = {
        "组成完整": "表内结构显示合计覆盖已识别分项；仅支持 L1 有限比较，指标定义与完整性仍待原文",
        "互不重叠": "表内结构显示分项为同一合计下的兄弟行/列；业务互斥仍待原文定义",
    }
    conditions = [_condition(name, reasons.get(name, "由表内结构自动登记；待指标级原文确认"), []) for name in re
quired]
    digest = hashlib.sha1(f"{total.occurrence_id}-{structure.get('layout')}".encode("utf-8")).hexdigest()[:10].u
pper()
    return RelationshipCheck(check_id=f"TABLE-{digest}", rule_id=rule_id, title=title[:160], topic=topic,
        operator=RULES[rule_id]["operator"],
        members=[_member(total, "total")] + [_member(row, "component") for row in components],
        conditions=conditions, relationship_evidence=[_reference(total)],
        relationship_basis="table_structure", table_structure=structure)


def _sum_values(rows: list[DisclosureOccurrence]) -> tuple[Decimal | None, str]:
    values, dimensions = [], set()
    for row in rows:
        value, dimension = _normalized(row)
        values.append(value)
        dimensions.add(dimension)
    if len(dimensions) != 1:
        return None, ""
    return sum(values, Decimal(0)), next(iter(dimensions))


def _column_candidates(table: dict) -> list[RelationshipCheck]:
    """同一行、不同列：合计列覆盖相邻分项列（如钢铁+其他=合计）。"""
    checks = []
    for row_index, columns in table["rows"].items():
        if len(columns) < 3:
            continue
        cells = [columns[index] for index in sorted(columns)]
        roles = {row.occurrence_id: _role_from_header(table["column_labels"].get(row.column_index, "")) for row 
in cells}
        totals = [row for row in cells if roles[row.occurrence_id] == "total" or row.column_role == "total"]
        components = [row for row in cells if row not in totals and (
            roles[row.occurrence_id] == "component" or row.column_role == "component")]
        if len(totals) != 1 or len(components) < 2:
            continue
        total = totals[0]
        components = sorted(components, key=lambda row: row.column_index)
        if total.column_index < max(row.column_index for row in components):
            continue
        selected_ids = {row.occurrence_id for row in components} | {total.occurrence_id}
        between = [row for column, row in columns.items()
                   if min(item.column_index for item in components) <= column < total.column_index]
        if any(row.occurrence_id not in selected_ids for row in between):
            # 合计与分项之间还有未识别的同表列，不能宣称覆盖全部。
            continue
        total_value, _ = _normalized(total)
        component_sum, _ = _sum_values(components)
        if component_sum is None:
            continue
        coverage = {"component_count": len(components),
                    "total_covers_columns": True,
                    "all_components_between_total": True,
                    "total_equals_sum": total_value == component_sum,
                    "sum_residual": text_number(total_value - component_sum),
                    "basis": "同一表格、同一指标行、合计列位于分项列右侧"}
        non_overlap = {"status": "structural_assumption",
                       "basis": "分项列为同一合计行下的兄弟列，坐标上互不重复",
                       "limitations": ["列结构不证明业务分类互斥，仍需原文定义"]}
        structure = _structure_payload(table["table_id"], "column_sum", total, components,
            coverage=coverage, non_overlap=non_overlap,
            limitations=["表内结构仅支持 L1 有限可比；指标定义、统计范围与期间仍需原文核实",
                         "合计列等于分项之和只证明算术相容，不证明企业披露正确"])
        rule_id = SUM_TOPICS.get(total.topic)
        if not rule_id:
            continue
        checks.append(_check(rule_id, f"{total.label[:60]} · 表内合计列与分项列", total.topic,
                             total, components, structure))
    return checks


def _previous_total_index(table: dict, row_index: int) -> int:
    previous = [index for index in table["rows"] if index < row_index
                and _role_from_label(table["row_labels"].get(index, "")) == "total"]
    return max(previous) if previous else -1


def _row_candidates(table: dict) -> list[RelationshipCheck]:
    """同一列、不同行：合计行覆盖其上分项行（如费用化+资本化=总投入）。"""
    checks = []
    by_column: dict[int, list[tuple[int, DisclosureOccurrence]]] = defaultdict(list)
    for row_index, columns in table["rows"].items():
        for column_index, row in columns.items():
            by_column[column_index].append((row_index, row))
    for column_index, items in by_column.items():
        items.sort(key=lambda pair: pair[0])
        for position, (row_index, total) in enumerate(items):
            if _role_from_label(total.label) != "total" and total.row_role != "total":
                continue
            if position == 0:
                continue
            lower = _previous_total_index(table, row_index)
            between = [row for index, row in items[position - 1::-1] if index > lower]
            components = [row for row in between
                          if total.document_id == row.document_id
                          and numeric_unit(row.unit)[0] == numeric_unit(total.unit)[0]]
            # 只接受紧邻合计行、带分项语义的行；普通明细不能凑数。
            components = [row for row in components if _role_from_label(row.label) == "component"][:12]
            if len(components) < 2:
                continue
            if any(_role_from_label(row.label) != "component" and row.value is not None for row in between
                   if row.occurrence_id not in {item.occurrence_id for item in components}):
                # 合计与分项之间还有未识别的同列数值，不能宣称覆盖完整。
                continue
            components.reverse()
            total_value, dimension = _normalized(total)
            component_sum, component_dimension = _sum_values(components)
            if component_dimension and component_dimension != dimension:
                continue
            coverage = {"component_count": len(components),
                        "total_covers_rows": True,
                        "all_components_between_total": True,
                        "total_equals_sum": component_sum is not None and total_value == component_sum,
                        "sum_residual": text_number(total_value - component_sum) if component_sum is not None el
se "",
                        "basis": "同一表格、同一数值列，合计行位于分项行之后"}
            non_overlap = {"status": "structural_assumption",
                           "basis": "分项行为同一合计行上方的兄弟行，坐标上互不重复",
                           "limitations": ["行结构不证明业务分类互斥，仍需原文定义"]}
            structure = _structure_payload(table["table_id"], "row_sum", total, components,
                coverage=coverage, non_overlap=non_overlap,
                limitations=["表内结构仅支持 L1 有限可比；指标定义、统计范围与期间仍需原文核实",
                             "分项加总等于合计只证明算术相容，不证明分项完整"])
            rule_id = SUM_TOPICS.get(total.topic)
            if not rule_id:
                continue
            checks.append(_check(rule_id, f"{total.label[:60]} · 表内合计行与分项行", total.topic,
                                 total, components, structure))
    return checks


def _cross_table_candidates(tables: dict[str, dict]) -> list[RelationshipCheck]:
    """跨表桥接只在同文档、相邻页、同期间且标签同源时生成；默认不启用。"""
    checks = []
    table_items = list(tables.values())
    for left in table_items:
        for right in table_items:
            if left is right or left["document_id"] != right["document_id"]:
                continue
            if abs(left["pdf_page"] - right["pdf_page"]) > 1:
                continue
            left_rows = [row for columns in left["rows"].values() for row in columns.values()]
            right_rows = [row for columns in right["rows"].values() for row in columns.values()]
            totals = [row for row in left_rows if _role_from_label(row.label) == "total"]
            components_pool = [row for row in right_rows if _role_from_label(row.label) == "component"]
            for left_total in totals:
                components = [row for row in components_pool
                              if row.year == left_total.year
                              and numeric_unit(row.unit)[0] == numeric_unit(left_total.unit)[0]]
                if len(components) < 2:
                    continue
                total_core = re.sub(r"(合计|总计|总额|小计)$", "", re.sub(r"[\s：:（）()]", "", left_total.label
))
                if not total_core or not all(total_core[:2] in row.label or row.label[:2] in total_core
                                             for row in components):
                    continue
                total_value, _ = _normalized(left_total)
                component_sum, _ = _sum_values(components)
                coverage = {"component_count": len(components),
                            "total_equals_sum": component_sum is not None and total_value == component_sum,
                            "sum_residual": text_number(total_value - component_sum) if component_sum is not Non
e else "",
                            "basis": "同文档相邻页、同期间列、标签同源的跨表桥接"}
                non_overlap = {"status": "unknown",
                               "basis": "跨表结构无法证明业务互斥",
                               "limitations": ["跨表桥接仅作线索，必须由原文定义确认"]}
                structure = _structure_payload(left["table_id"], "cross_table", left_total, components,
                    coverage=coverage, non_overlap=non_overlap,
                    limitations=["跨表桥接只支持 L1 线索；表间对应关系仍需原文说明",
                                 "未确认两张表是否同一统计范围与期间"])
                rule_id = SUM_TOPICS.get(left_total.topic)
                if not rule_id:
                    continue
                checks.append(_check(rule_id, f"{left_total.label[:60]} · 跨表合计与分项", left_total.topic,
                                     left_total, components, structure))
    return checks


def build_table_composition_candidates(occurrences: list[DisclosureOccurrence], period: str, *,
                                       include_cross_table: bool = False) -> list[RelationshipCheck]:
    """返回表内合计/分项候选；默认只做高置信同表结构。

    返回的 check_id 以 TABLE- 开头，成员、条件和结构证据可直接序列化。
    调用方仍须执行 `upgrade_conditions` 与 `calculate`，不能把结构候选
    直接计为企业异常或正式结论。
    """
    tables = _rows_by_table(occurrences, period)
    checks = []
    for table in tables.values():
        checks.extend(_column_candidates(table))
        checks.extend(_row_candidates(table))
    if include_cross_table:
        checks.extend(_cross_table_candidates(tables))
    # 同一合计/分项集合只保留一次，避免列与行两种读法重复计数。
    unique, seen = [], set()
    for check in checks:
        signature = (check.rule_id, tuple(sorted((member.occurrence_id, member.role) for member in check.members
)))
        if signature in seen:
            continue
        seen.add(signature)
        unique.append(check)
    return unique
"""跨年度比较：只有同企业往年报告已登记且原件可用时才开放，否则显式缺口。

专业边界（必须同时成立）：
- 未登记往年报告时返回明确缺口，不推定"同比改善/恶化"，也不伪造往年数值；
- 两侧指标必须通过 metric_identity 严格配对（同口径、同单位），单位不同拒绝换算比较；
- 比较结果只覆盖该指标本身，不外推到企业整体表现；
- 每一侧都必须带 document_id 与 pdf_page，缺页码的结果不得作为证据。
"""
from __future__ import annotations

from decimal import Decimal, InvalidOperation

from .connectivity_metrics import metric_identity

VERSION = "connectivity_temporal_v1"


def temporal_status(run, root=None) -> dict:
    """本次运行登记了哪些报告期间；少于两个期间则跨年比较不可用。"""
    periods = sorted({(doc.report_period or doc.period, doc.company)
                      for doc in getattr(run, "documents", [])})
    companies = {company for _, company in periods}
    unique_periods = sorted({period for period, _ in periods})
    available = len(unique_periods) >= 2 and len(companies) == 1
    registered_eligible = available
    source_verified = False
    if available and root is not None:
        from .connectivity_extraction import source_path
        import hashlib
        source_verified = all(source_path(root, doc).is_file() and
            hashlib.sha256(source_path(root, doc).read_bytes()).hexdigest() == doc.file_sha256
            for doc in run.documents)
    available = available and source_verified
    return {
        "version": VERSION,
        "available": available,
        "registered_eligible": registered_eligible,
        "source_verified": source_verified,
        "registered_periods": unique_periods,
        "companies": sorted(companies),
        "reason": "" if available else ("跨年原件尚未核验或不可读取；不开放数值比较。" if registered_eligible el
se
        "本次运行未登记同企业其他年度报告；跨年比较不可自动执行，需先登记往年原件。"),
    }


def _decimal(value):
    try:
        number = Decimal(str(value))
        return number if number.is_finite() else None
    except (InvalidOperation, TypeError, ValueError):
        return None


def compare_periods(current: dict, prior: dict) -> dict:
    """对同一指标的两个已登记期间值做确定性比较；两侧证据身份必须完整。

    输入同时包含指标、数值、单位、文件编号、页码和年度。
    比较身份还须声明公司、组织范围、指标定义、分母和可比范围。
    同一年度的两个数不能包装成跨年增长率。
    当前年度必须晚于基期，不自动猜测输入顺序。
    重述或口径变化先阻断直接比较，不能靠加免责声明放行。
    原件核验身份必须由上游登记流程提供，不能从页码存在反推。
    缺少必要身份时返回资格矩阵，不编造可比结论。
    有效计算仍只描述该指标，不能外推为公司整体表现。
    """
    for side, item in (("current", current), ("prior", prior)):
        if not item.get("document_id") or not item.get("pdf_page"):
            return {"version": VERSION, "available": False,
                    "reason": f"{side} 侧缺少 document_id 或 pdf_page；没有页码的比较不作为证据。"}
    current_identity = metric_identity(current.get("label", ""))
    prior_identity = metric_identity(prior.get("label", ""))
    if not current_identity["metric_id"] or current_identity["metric_id"] != prior_identity["metric_id"]:
        return {"version": VERSION, "available": False,
                "reason": "两侧指标未通过严格口径配对（metric_identity 不一致）；拒绝跨口径同比。"}
    if str(current.get("unit") or "") != str(prior.get("unit") or ""):
        return {"version": VERSION, "available": False,
                "reason": f"单位不一致（{current.get('unit')} vs {prior.get('unit')}）；程序不做单位换算比较。"}
    # 同年不同数值不是同比；重述和合并范围变化必须先取得可比重编依据。
    import re
    years = [str(item.get("period", "")) for item in (current, prior)]
    if not all(re.fullmatch(r"20\d{2}", year) for year in years) or int(years[0]) <= int(years[1]):
        return {"version": VERSION, "available": False, "reason": "比较期间必须为明确的前后年度，当前年须晚于基
期。"}
    dimensions = ("company", "scope", "metric_definition", "denominator", "comparable_range")
    gaps = [name for name in dimensions if not current.get(name) or not prior.get(name) or current[name] != prio
r[name]]
    if gaps or any(item.get("restated") or item.get("definition_changed") for item in (current, prior)):
        return {"version": VERSION, "available": False, "reason": "跨年可比前提缺失或冲突：" + "、".join(gaps or
 ["重述/口径变化"]),
                "qualification": {name: {"current": current.get(name), "prior": prior.get(name)} for name in dim
ensions}}
    if not all(item.get("source_verified") is True for item in (current, prior)):
        return {"version": VERSION, "available": False, "reason": "两侧原件及页码尚未核验，不开放同比。"}
    current_value = _decimal(current.get("value"))
    prior_value = _decimal(prior.get("value"))
    if current_value is None or prior_value is None:
        return {"version": VERSION, "available": False, "reason": "数值无法转为十进制。"}
    if prior_value == 0:
        return {"version": VERSION, "available": False, "reason": "往年基数为0；同比比例无定义。"}
    difference = current_value - prior_value
    percent_change = difference / abs(prior_value) * Decimal(100)
    direction = "上升" if difference > 0 else ("下降" if difference < 0 else "持平")
    return {
        "version": VERSION,
        "available": True,
        "metric": current_identity["canonical"] or current.get("label", ""),
        "metric_id": current_identity["metric_id"],
        "unit": current.get("unit"),
        "current": {"period": current.get("period", ""), "value": str(current_value),
                    "document_id": current["document_id"], "pdf_page": current["pdf_page"]},
        "prior": {"period": prior.get("period", ""), "value": str(prior_value),
                  "document_id": prior["document_id"], "pdf_page": prior["pdf_page"]},
        "difference": str(difference),
        "percent_change": f"{percent_change:.4f}%",
        "direction": direction,
        "limitations": [
            "比较只覆盖该指标本身，不外推到企业整体表现",
            "两侧期间口径须与报告披露一致；程序未核对会计政策变更",
            "方向词只描述数值变化，不构成业绩评价或预测",
        ],
    }


def comparison_summary(root, run) -> dict:
    """展示登记资格和行业覆盖，缺少逐项依据时返回缺口矩阵。

    公司和期间来自登记原件元数据，不由模型补出。
    缓存文件数量只说明本机来源覆盖，不表示行业模型效果。
    跨年原件可读之后仍须通过逐指标的可比性检查。
    同业样本未满足完整身份时保留拒绝排名的明确说明。
    界面与导出消费同一份矩阵，避免各写一套样本数量。
    """
    from types import SimpleNamespace
    from .connectivity_extraction import load_registry, source_path
    from .connectivity_financial_bridge import industry_for, INDUSTRY_PROFILES
    from .connectivity_peer_benchmark import load_registry as peer_sources
    documents = load_registry(root)
    grouped = {}
    for doc in documents:
        industry = industry_for(doc.company)
        key = (doc.company, doc.period)
        row = grouped.setdefault(key, {"company": doc.company, "period": doc.period,
            "industry": INDUSTRY_PROFILES[industry]["name"], "documents": [], "cached_documents": 0})
        row["documents"].append({"document_id": doc.document_id, "report_type": doc.report_type,
                                 "source_url": doc.source_url, "source_sha256": doc.file_sha256})
        row["cached_documents"] += int(source_path(root, doc).is_file())
    temporal = temporal_status(SimpleNamespace(documents=[doc for doc in documents if doc.company == run.company
]), root=root)
    peers = peer_sources()
    return {"temporal": temporal, "industry_sources": list(grouped.values()),
            "peer": {"available": False, "registered_samples": len(peers.get("samples", [])),
                     "reason": "未提供通过逐指标期间、范围、定义、分母和来源核验的同业样本，不生成排名。"},
            "notice": "登记覆盖不等于行业效果验证；跨年/同业数值须另外通过逐指标资格门禁。"}
"""受控Agent工具注册表：模型只能申请白名单内工具，服务端验证并执行。

安全边界（必须同时成立）：
- 不接受任意网址、任意磁盘路径、可执行代码或未登记文档；
- document_id 必须出现在本次运行已登记文档里，页码必须落在原件范围内；

class ToolRequest:
    """一次工具申请：工具名与参数结构；参数由服务端验证，模型不能直接执行。"""

    def __init__(self, tool_name: str, arguments: dict):
        self.tool_name = str(tool_name or "").strip()
        self.arguments = dict(arguments or {})

    def validated_args_hash(self) -> str:
        payload = json.dumps({"tool": self.tool_name, "args": self.arguments},
                             ensure_ascii=False, sort_keys=True)
        return hashlib.sha256(payload.encode()).hexdigest()[:16]


def _digest(text: str) -> str:
    return hashlib.sha256(str(text or "").encode()).hexdigest()[:16]


def _find_document(run, document_id: str):
    wanted = str(document_id or "").strip()
    for doc in getattr(run, "documents", []):
        if doc.document_id == wanted:
            return doc
    return None


def _load_page_text(root, doc, page_no: int) -> str | None:
    """打开已登记原件的指定页；页码越界返回 None，不做任何下载。"""
    try:
        from .connectivity_extraction import source_path
        import pymupdf
        path = source_path(root, doc)
        with PDF_ACCESS_LOCK, pymupdf.open(path) as pdf:
            if page_no < 1 or page_no > len(pdf):
                return None
            return pdf[page_no - 1].get_text("text")
    except Exception:
        return None


def _validate_common(run, arguments: dict) -> tuple[None, None] | tuple[str, str]:
    """通用参数验证；返回 (错误码, 说明)。文档必须已登记。"""
    doc = _find_document(run, arguments.get("document_id"))
    if doc is None:
        return "UNREGISTERED_DOCUMENT", "document_id 不在本次已登记文档中；工具拒绝执行。"
    return None, None

               "status": "rejected", "output_digest": "", "source_hash": "",
               "elapsed_ms": 0, "result": None, "reason": ""}
    if request.tool_name not in TOOL_NAMES:
        receipt["reason"] = f"工具不在白名单：{request.tool_name}"
        return receipt
    if getattr(run, "versions", {}).get("experiment_mode") == "fixed_evidence" and request.tool_name in {"search

        return _finish(receipt, started)

            receipt["reason"]=str(error)[:350]
            return _finish(receipt,started)
        receipt.update(status="hit",result=result,output_digest=_digest(json.dumps(result,ensure_ascii=False)))
        return _finish(receipt,started)
    if request.tool_name == "search_matter_reports":
        from .connectivity_matter_extraction import read_corpus, search_corpus
        query = str(args.get("query") or "").strip()
        ids = args.get("document_ids") or []
        if not query or len(query) > 300 or not isinstance(ids, list) or not set(ids) <= {d.document_id for d in
 run.documents}:
            receipt["reason"] = "查询或登记文档范围无效"
            return _finish(receipt, started)
        # 仍验证实际字节，不能以已缓存证据绕过原件换版。
        from .connectivity_extraction import ensure_source
        for doc in run.documents:
            if not ids or doc.document_id in ids:
                ensure_source(root, doc)
        if corpus is None:
            corpus, _ = read_corpus(root, run.documents)
        result = search_corpus(corpus, query, document_ids=ids, report_type=args.get("report_type"),
            periods=args.get("periods"), limit=max(1, min(12, int(args.get("limit", 8)))))
        receipt.update(status="hit" if result["evidence"] else "no_match", result=result,
            output_digest=_digest(json.dumps(result, ensure_ascii=False)),
            source_hash=_digest("|".join(d.file_sha256 for d in run.documents)))
        return _finish(receipt, started)
    error = None
    if request.tool_name in {"search_registered_reports", "open_source_page"}:
        error, reason = _validate_common(run, args)
        if error:
            receipt["reason"] = reason
            return _finish(receipt, started)
        if root is not None:
            # 生产入口核实原件，不让旧登记哈希冒充换版后的文件内容。
            from .connectivity_extraction import source_path
            doc = _find_document(run, args.get("document_id"))
            path = source_path(root, doc)
            if not path.is_file() or hashlib.sha256(path.read_bytes()).hexdigest() != doc.file_sha256:
                receipt["reason"] = "来源原件缺失或哈希已变化；本次工具拒绝使用失效证据。"
                return _finish(receipt, started)
    if request.tool_name == "search_registered_reports":
        return _search_registered_reports(root, run, args, receipt, started)
    if request.tool_name == "open_source_page":
        receipt=_open_source_page(root, run, args, receipt, started)
        if getattr(run,"matter_schema_version", "") and receipt["status"]=="hit":
            from .connectivity_matter_extraction import read_corpus
            if corpus is None:corpus,_=read_corpus(root,[_find_document(run,args.get("document_id"))])
            receipt["result"]["evidence"]=[e.model_dump() for e in corpus if e.document_id==args["document_id"] 
and e.pdf_page==int(args["page"])]
            receipt["output_digest"]=_digest(json.dumps(receipt["result"],ensure_ascii=False))
        return receipt
    if request.tool_name == "lookup_ontology_rule":
        if getattr(run, "versions", {}).get("ablation") == "no_ontology_gate":
            receipt["reason"] = "实验消融条件：当前运行已关闭会计语义本体规则库查询"
            return _finish(receipt, started)
        return _lookup_ontology_rule(args, receipt, started)
    if request.tool_name == "recompute_decimal":
        return _recompute_decimal(args, receipt, started)
    if request.tool_name == "lookup_annual_comparison":
        return _lookup_annual_comparison(root, run, args, receipt, started)
    return _lookup_peer_sample(run, args, receipt, started)


def _finish(receipt: dict, started: float) -> dict:
    receipt["elapsed_ms"] = round((time.perf_counter() - started) * 1000)
    return receipt


def _search_registered_reports(root, run, args, receipt, started):
    """在已登记报告内检索关键词；不接受外部网络或路径。"""
    terms = [str(term).strip() for term in (args.get("terms") or []) if str(term).strip()]
    doc = _find_document(run, args.get("document_id"))
    if not terms:
        receipt["reason"] = "缺少检索词。"
        return _finish(receipt, started)
    if len(terms) > 5:
        terms = terms[:5]
    from .connectivity_extraction import source_path
    import pymupdf
    hits = []
    try:
        path = source_path(root, doc)
        with PDF_ACCESS_LOCK, pymupdf.open(path) as pdf:
            receipt["source_hash"] = doc.file_sha256 or _digest(str(path))
            for index, page in enumerate(pdf):
                if time.perf_counter() - started > 10:
                    receipt["reason"] = "工具达到10秒检索上限；仅返回已检查范围，不推定未披露。"
                    break
                text = page.get_text("text")
                if not text:
                    continue
                matched = [term for term in terms if term in text]
                if matched:
                    anchor = min(text.find(term) for term in matched)
                    snippet = text[max(0, anchor - 120):anchor + 600].replace("\n", " ")
                    hits.append({"page": index + 1, "matched_terms": matched, "snippet": snippet})
                if len(hits) >= 8:
                    break
    except Exception as error:
        receipt["reason"] = f"检索失败：{type(error).__name__}"
        return _finish(receipt, started)
    receipt.update(status="hit" if hits else "no_match", result={"document_id": doc.document_id,
                   "hits": hits, "terms": terms, "quote": "\n".join(hit["snippet"] for hit in hits),
                   "matched_terms": sorted({term for hit in hits for term in hit["matched_terms"]}),
                   "evidence_refs": [{"document_id": doc.document_id, "pdf_page": hit["page"],
                                      "source_sha256": receipt["source_hash"], "quote": hit["snippet"]} for hit 
in hits]})
    receipt["output_digest"] = _digest(json.dumps(hits, ensure_ascii=False))
    if not hits:
        receipt["reason"] = "已检索范围内未命中；不推定企业未披露。"
    return _finish(receipt, started)


def _open_source_page(root, run, args, receipt, started):
    """打开登记原件指定页并返回原文摘录；页码必须存在。"""
    doc = _find_document(run, args.get("document_id"))
    try:
        page_no = int(args.get("page"))
    except (TypeError, ValueError):
        receipt["reason"] = "缺少有效页码。"
        return _finish(receipt, started)
    text = _load_page_text(root, doc, page_no)
    if text is None:
        receipt["reason"] = "页码超出原件范围或原件不可读。"
        return _finish(receipt, started)
    receipt["source_hash"] = doc.file_sha256 or _digest(str(doc.source_url))
    terms = [str(term) for term in args.get("terms", [])][:5]
    matched = [term for term in terms if term in text]
    anchor = min((text.find(term) for term in matched), default=0)
    quote = text[max(0, anchor - 120):anchor + 900].replace("\n", " ")
    receipt.update(status="hit", result={"document_id": doc.document_id, "page": page_no,
                                         "quote": quote, "matched_terms": matched,
                                         "evidence_refs": [{"document_id": doc.document_id, "pdf_page": page_no,
                                                            "source_sha256": receipt["source_hash"], "quote": qu
ote}]})
    receipt["output_digest"] = _digest(quote)
    return _finish(receipt, started)

def _recompute_decimal(args, receipt, started):
    """确定性十进制复算：只允许两操作数与受控算子，不执行任意表达式。"""
    from decimal import Decimal, InvalidOperation
    operator = str(args.get("operator") or "").strip()
    if operator not in {"sum", "difference", "ratio", "percent_change"}:
        receipt["reason"] = "算子不在白名单（sum/difference/ratio/percent_change）。"
        return _finish(receipt, started)
    try:
        left = Decimal(str(args.get("left")))
        right = Decimal(str(args.get("right")))
    except (InvalidOperation, TypeError, ValueError):
        receipt["reason"] = "操作数不是有效数值。"
        return _finish(receipt, started)
    if operator == "ratio" and right == 0:
        receipt["reason"] = "比率的分母为0；拒绝计算。"
        return _finish(receipt, started)
    if operator == "sum":
        value = left + right
        formula = f"{left} + {right} = {value}"
    elif operator == "difference":
        value = left - right
        formula = f"{left} - {right} = {value}"
    elif operator == "ratio":
        value = left / right
        formula = f"{left} ÷ {right} = {value}"
    else:
        if right == 0:
            receipt["reason"] = "同比基数为0；拒绝计算。"
            return _finish(receipt, started)
        value = (left - right) / abs(right) * Decimal(100)
        formula = f"({left} - {right}) ÷ |{right}| × 100 = {value}%"
    receipt.update(status="hit", result={"operator": operator, "value": str(value),
                                         "formula": formula,
                                         "note": "确定性复算只核对该算式本身；不改变任何条件门禁状态。"})
    receipt["output_digest"] = _digest(formula)
    return _finish(receipt, started)

"""企业披露版本链：原文档、更正文档与生效关系独立登记。

版本变化只生成版本证据链，不进入正式跨报告差额，也不与系统修订混成一条时间轴。
该模块不依赖双报告运行：同一公司同一报告类型的多份登记文档即可构成版本链。
"""
from __future__ import annotations

import re

from .connectivity_engine import number
from .connectivity_schemas import ReportDocument, VersionChange, VersionRecord

VERSION_NOTE = "版本金额变化只说明财务披露版本发生变化，不能计为已解释的跨报告差异。"


def version_records_for(documents: list[ReportDocument]) -> list[VersionRecord]:
    """按披露日期排序列出各版本；媒体引述只记来源类别。"""
    ordered = sorted(documents, key=lambda doc: (doc.disclosure_date or "9999", doc.document_id))
    records = []
    for index, doc in enumerate(ordered, start=1):
        records.append(VersionRecord(version_id=f"{doc.document_id}-v{index}", document_id=doc.document_id,
            report_type=doc.report_type, version_role=doc.version_role, disclosure_date=doc.disclosure_date, met
ric_period=doc.metric_period,
            report_period=doc.report_period or doc.period,
            label=f"{doc.company} {doc.report_type} {doc.report_period or doc.period}",
            effective_relation=("首次公开版本" if doc.version_role == "original" else
                                "更正后生效" if doc.version_role == "correction" else "版本角色未登记"),
            source_kind="mirror" if doc.source_kind == "announcement_mirror" else "official"))
    return records


def _key(row, report_type: str) -> tuple[str, str, str, str]:
    """同一指标必须在同一报告类型内比较；AR 与 ESG 不是彼此的版本。"""
    return (report_type, re.sub(r"\s+", "", row.label), row.year, row.unit)


def build_version_chain(entries: list[dict]) -> dict:
    """entries 为 [{'document': ReportDocument, 'occurrences': [...]}]；返回记录、变化与限制说明。"""
    documents = [entry["document"] for entry in entries]
    records = version_records_for(documents)
    ordered = sorted(entries, key=lambda entry: (entry["document"].report_type,
                                                 entry["document"].disclosure_date or "9999",
                                                 entry["document"].document_id))
    # 逐版比较：前一版本 → 本版本，形成 v1→v2→v3 的链，而不是每个更正都对原始版本比。
    latest: dict[tuple, tuple] = {}
    changes: list[VersionChange] = []
    for entry in ordered:
        doc = entry["document"]
        for row in entry.get("occurrences", []):
            if not row.value:
                continue
            key = _key(row, doc.report_type)
            base = latest.get(key)
            if base and doc.version_role == "correction":
                base_doc, base_row = base
                try:
                    difference = number(row.value) - number(base_row.value)
                except ValueError:
                    difference = None
                if difference is not None:
                    changes.append(VersionChange(from_version_id=f"{base_doc.document_id}-{base_row.occurrence_i
d}",
                        to_version_id=f"{doc.document_id}-{row.occurrence_id}", label=row.label,
                        display_change=f"{difference}", absolute_change=f"{abs(difference)}", unit=row.unit,
                        only_version_evidence=True))
            latest[key] = (doc, row)
    return {"records": records, "changes": changes,
            "note": VERSION_NOTE,
            "limitation": ("本次未登记更正版本，版本链只有原始版本" if not changes else
                           "变化只覆盖同一标签、同一年度、同一单位的行；未匹配行不推断差值"),
            "source_kind_rule": "official 可形成版本链；mirror 单独标注；媒体引述不生成文件身份或页码"}
