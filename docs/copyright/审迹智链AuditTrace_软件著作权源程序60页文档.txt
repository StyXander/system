@app.middleware('http')
async def connectivity_timing(request, call_next):
    started = time.perf_counter()
    response = await call_next(request)
    if request.url.path.startswith('/api/connectivity/runs'):
        duration = (time.perf_counter() - started) * 1000
        response.headers['Server-Timing'] = f'app;dur={duration:.2f}'
        logging.getLogger('uvicorn.error').info('connectivity_http method=%s path=%s status=%s duration_ms=%.2f bytes=%s',
            request.method, request.url.path, response.status_code, duration, response.headers.get('content-length', 'stream'))
    return response

def build_router(root: Path, reserve_model, quota_ledger) -> APIRouter:
    """延迟绑定现有应用配额，避免循环导入和第二套模型预算。"""
    router = APIRouter(prefix="/api/connectivity", tags=["connectivity"])
    lock = threading.RLock()
    stores = {}
    directory = root / "backend/runtime/connectivity"
    directory.mkdir(parents=True, exist_ok=True)
    result_cache = OrderedDict()
    log = logging.getLogger(__name__)

    def normalized_result(item, *, layer="full"):
        body = item.get("result")
        if not body or body.get("projection_supported") is False:
            return body
        from .connectivity_extraction import source_path
        from .connectivity_source_cache import source_signature
        registry = {d.document_id: d for d in load_registry(root)}
        signatures = []
        for original in body.get("documents", []):
            doc = registry.get(original["document_id"])
            try:
                signatures.append((original["document_id"], doc.file_sha256, source_signature(source_path(root, doc))))
            except (OSError, AttributeError, ValueError):
                signatures.append((original["document_id"], "missing"))
        rules = root / "backend/app/connectivity_matter_rules.json"
        try:
            rule_signature = source_signature(rules)
        except OSError:
            rule_signature = None
        version = item.get("version", item.get("updated_at"))
        key = (item["task_id"], version, layer, tuple(signatures), rule_signature)


// ==============================================================================
// 审迹智链跨报告证据链追踪与可持续信息智能鉴证系统 V1.0 - 第 4 页 / 共 60 页
// ==============================================================================
        with lock:
            if key in result_cache:
                result_cache.move_to_end(key)
                log.info("connectivity_result cache=hit task=%s layer=%s version=%s", item["task_id"], layer, version)
                return deepcopy(result_cache[key])
        normalized = ConnectivityRun.model_validate({k: v for k, v in body.items() if k in ConnectivityRun.model_fields})
        normalized = refresh_run_sources(root, normalized)
        if layer == "full":
            if normalized.matter_schema_version:
                from .connectivity_assurance_planning import build_assurance_planning
                normalized.assurance_planning = build_assurance_planning(normalized)
            value = normalized.model_dump(mode="json")
            value["overview"] = build_overview(normalized)
        else:
            all_fields = normalized.model_dump(mode="json")
            value = {k: all_fields.get(k, v) for k, v in body.items()}
            value["versions"] = all_fields["versions"]
            if layer in {"summary", "matters"} and normalized.versions.get("source_validity", {}).get("status") == "invalid":
                # 摘要没有证据全文，不能继续展示来源有效的旧判断。
                value["assurance_planning"] = {}
                for assessment in value.get("matter_assessments", []):
                    assessment["relation_support"] = "invalid"
                    assessment["unresolved"] = [*assessment.get("unresolved", []), "依赖来源变化，打开事项查看失效证据"]
        value["ai_generated_content_notice"] = NOTICE
        with lock:
            # 版本变化即淘汰旧版本；来源变化也不积累大结果。
            for old in list(result_cache):
                if old[0] == item["task_id"] and (old[1] != version or old[2] == layer):
                    del result_cache[old]
            result_cache[key] = deepcopy(value)
            while len(result_cache) > 16:
                result_cache.popitem(last=False)
        log.info("connectivity_result cache=miss task=%s layer=%s version=%s", item["task_id"], layer, version)
        return value

    def store(task_id=None):
        mode = "local" if task_id and task_id.startswith("CONNJOB-") else ("supabase" if task_id or demo_task_supabase_enabled() else "local")
        with lock:
            if mode not in stores:
                stores[mode] = SupabaseDemoRunTaskStore(get_demo_task_client()) if mode == "supabase" else DemoRunTaskStore(directory / "tasks", task_id_prefix="CONNJOB")
        return stores[mode]

    def owner(request: Request) -> str:
        return hashlib.sha256(request.cookies.get(COOKIE, "").encode()).hexdigest()

    def task(task_id: str):

        if not result or result.get("run_body", {}).get("kind") != "connectivity":
            raise HTTPException(404, "关联核对任务不存在")
        return result

    def authorize(request: Request, item: dict):
        if not single_user_enabled() and (not request.cookies.get(COOKIE) or not hmac.compare_digest(owner(request), item["run_body"].get("owner", ""))):
            raise HTTPException(403, "只能更正或取消当前浏览器创建的任务")
        origin = request.headers.get("origin")

            raise HTTPException(403, "更正请求必须来自当前网站")

    def public(item: dict, request: Request | None = None) -> dict:
        view = store(item["task_id"]).snapshot(item["task_id"], item=item)
        if not view:
            raise HTTPException(404, "任务不存在")
        result = {k: v for k, v in view.items() if k not in {"run_body", "request_payload", "request_sha256", "idempotency_key_sha256"}}
        result["parent_task_id"] = item.get("run_body", {}).get("parent_task_id")
        is_owner = single_user_enabled() or bool(request and request.cookies.get(COOKIE)
                        and hmac.compare_digest(owner(request), item.get("run_body", {}).get("owner", "")))
        if item.get("run_body", {}).get("analysis_scope") == "matter_connectivity" and not is_owner:
            raise HTTPException(403, "事项调查任务仅当前创建会话可读取")
        result["can_modify"] = is_owner
        # 公共视图默认不含个人观察正文，只列条数；本人视图才展开。
        body = result.get("result")
        if isinstance(body, dict):
            body = normalized_result({**item, "result": body})
            result['result'] = body
        if isinstance(body, dict) and not is_owner:
            body = dict(body)
            notes = body.get("professional_notes") or []
            body["personal_observation_count"] = len(notes)
            body["professional_notes"] = []
            result["result"] = body
        elif isinstance(body, dict):
            body = dict(body)
            body["personal_observation_count"] = len(body.get("professional_notes") or [])
            result["result"] = body
        return result

def _divide_interval(numerator: Interval, denominator: Interval) -> Interval | None:
    """有界区间相除；极值的开闭来自实际达到该极值的端点组合。"""
    if denominator.low <= 0 <= denominator.high:
        return None
    corners = [(n / d, nc and dc)
               for n, nc in ((numerator.low, numerator.low_closed), (numerator.high, numerator.high_closed))
               for d, dc in ((denominator.low, denominator.low_closed), (denominator.high, denominator.high_closed))]
    low, high = min(value for value, _ in corners), max(value for value, _ in corners)
    return Interval(low, high, any(closed for value, closed in corners if value == low),
                    any(closed for value, closed in corners if value == high))

def member_boundary(occurrence: DisclosureOccurrence) -> str:
    """报告边界优先取显式字段；表内分组和报告级声明不能替代合并/母公司判断。"""
    if occurrence.report_boundary != "unknown":
        return occurrence.report_boundary
    if occurrence.scope_level in {"report", "table"}:
        return "unknown"
    scope = re.sub(r"\s+", "", occurrence.scope or "")
    if "母公司" in scope:
        return "parent"
    if re.search(r"合并(?:报表|范围|口径|数据)", scope) or scope == "合并":
        return "consolidated"
    if re.search(r"分部|板块", scope):
        return "segment"
    return "unknown"

def boundary_comparison(check: RelationshipCheck, occurrences: list[DisclosureOccurrence]) -> dict:
    """同值关系先比较主体边界；不同口径只说明不可直接比较，不认定披露错误。"""
    found = {row.occurrence_id: row for row in occurrences}
    rows = [found[member.occurrence_id] for member in check.members if member.occurrence_id in found]
    if check.operator != "equal" or len(rows) < 2:
        return {"status": "not_applicable", "message": ""}
    boundaries = {row.occurrence_id: member_boundary(row) for row in rows}
    known = {value for value in boundaries.values() if value != "unknown"}
    table_scopes = {row.table_scope.strip() for row in rows if row.table_scope.strip()}
    status = "unknown"
    message = ""
    if len(known) > 1:
        status = "different"
        message = "合并报表与母公司报表属于不同口径，不能直接比较；不认定为企业披露错误"
    elif len(known) == 1 and not all(value != "unknown" for value in boundaries.values()):
        # 一侧明确是母公司/合并而另一侧边界未证实，同样不能直接比较；
        # 这是证据不足，不是把未知一侧默认为企业披露错误。
        status = "undetermined"
        known_value = next(iter(known))
        label = {"parent": "母公司", "consolidated": "合并", "segment": "分部"}.get(known_value, known_value)
        message = f"一侧为{label}报表口径，另一侧报告边界未证实；不能直接比较（不认定为企业披露错误）"
    elif len(known) == 1 and all(value != "unknown" for value in boundaries.values()):
        status = "same"
    if len(table_scopes) > 1:
        status = "different"
        message = (message + "；" if message else "") + "成员来自不同表内分组，不能当作同一指标口径"
    return {"status": status, "boundaries": boundaries, "table_scopes": sorted(table_scopes),
            "message": message, "note": "报告边界相同不证明指标统计范围相同"}

def validate_refs(refs: list[EvidenceRef], occurrences: dict[str, DisclosureOccurrence]) -> bool:
    """引用必须属于本次报告且逐字存在于对应上下文，不接受模型编造出处。"""
    compact = lambda s: re.sub(r"\s+", "", s)
    return bool(refs) and all(
        ref.occurrence_id in occurrences
        and len(compact(ref.quote)) >= 2  # 空白引文不能成为任何口径条件的证据
        and compact(ref.quote) in compact(occurrences[ref.occurrence_id].context)
        for ref in refs
    )

def _trial_calculation(trial: dict) -> dict:
    """假设试算独立存放，不占用正式差额字段，也不进入已衔接统计。"""
    return {"hypothetical": True,

            "expression": trial["expression"], "residual": trial["residual"],
            "residual_interval": _residual_interval(trial).as_dict(),
            "left_interval": trial.get("left_interval"), "right_interval": trial.get("right_interval"),
            "positions": trial.get("positions", []), "normalized_unit": trial.get("normalized_unit", ""),
            "denominator_check": trial.get("denominator_check")}

def _residual_interval(trial: dict) -> Interval:
    """差额区间由两侧区间相减得到；多点位关系取各位置区间的包络，不伪造精确值。"""
    positions = trial.get("positions") or []
    if positions:
        intervals = [item.get("residual_interval") for item in positions if item.get("residual_interval")]
        if not intervals:
            # 无区间数据时标记为未知，不用点值冒充精确区间。
            value = Decimal(trial["residual"])
            return Interval(value, value, False, False)
        lows = [Decimal(item["low"]) for item in intervals]
        highs = [Decimal(item["high"]) for item in intervals]
        closed = all(item.get("low_closed") and item.get("high_closed") for item in intervals)
        return Interval(min(lows), max(highs), closed, closed)
    left = trial.get("left_interval") or {}
    right = trial.get("right_interval") or {}
    if not left or not right:
        value = Decimal(trial["residual"])
        return Interval(value, value, False, False)
    return Interval(Decimal(left["low"]), Decimal(left["high"]), left["low_closed"], left["high_closed"]).minus(
        Interval(Decimal(right["low"]), Decimal(right["high"]), right["low_closed"], right["high_closed"]))

def _amount_interval(amount: Decimal, factor: Decimal, rounding: str) -> tuple[Interval, str]:
    """调节额自身的精度区间：未声明舍入方法时不伪造区间。"""
    value = amount * factor
    if rounding == "exact":
        return Interval(value, value), "exact"
    if rounding == "half_up":
        precision = Decimal(1).scaleb(amount.as_tuple().exponent) * factor / 2
        return Interval(value - precision, value + precision), "known"
    return Interval(value, value, False, False), "unknown"

def _all_refs(value):
    if isinstance(value,dict):
        for k,v in value.items():
            if k.endswith("evidence_ids") and isinstance(v,list):yield from v

        setattr(run,field,getattr(working,field))

def _word_text(value) -> str:
    """PDF可能含XML非法控制字符；仅净化Word显示，不改JSON中的原始证据。"""
    return re.sub(r"[\x00-\x08\x0b\x0c\x0e-\x1f\ud800-\udfff\ufffe\uffff]", "", str(value))
